cmake_minimum_required(VERSION 3.18)
project(catrobot_native LANGUAGES CXX)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_native native/bindings.cpp)
target_compile_features(_native PRIVATE cxx_std_17)
target_compile_options(_native PRIVATE -Wall -Wextra -Werror)
target_include_directories(_native PRIVATE ../cat-robot-wiring-visual/sketches/cat_robot)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../cat-robot-wiring-visual/sketches/cat_robot/control.h" SAFETY_SHA)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../cat-robot-wiring-visual/sketches/cat_robot/drive.h" DRIVE_SHA)
string(SHA256 SOURCE_SHA "${SAFETY_SHA}${DRIVE_SHA}")
target_compile_definitions(_native PRIVATE CONTROL_SOURCE_SHA="${SOURCE_SHA}")
install(TARGETS _native DESTINATION catrobot)
